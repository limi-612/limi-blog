## 错误监控体系

### 全局错误捕获
```javascript
// JavaScript运行时错误
window.addEventListener('error', (event) => {
    const errorInfo = {
        message: event.message,
        filename: event.filename,
        lineno: event.lineno,
        colno: event.colno,
        stack: event.error?.stack,
        timestamp: Date.now()
    };
    
    reportError(errorInfo);
});

// Promise未捕获错误
window.addEventListener('unhandledrejection', (event) => {
    const errorInfo = {
        type: 'unhandledrejection',
        reason: event.reason,
        timestamp: Date.now()
    };
    
    reportError(errorInfo);
});
```

### 资源加载错误
```javascript
// 监听资源加载失败
window.addEventListener('error', (event) => {
    const target = event.target;
    
    if (target !== window) {
        const errorInfo = {
            type: 'resource',
            tagName: target.tagName,
            src: target.src || target.href,
            message: '资源加载失败',
            timestamp: Date.now()
        };
        
        reportError(errorInfo);
    }
}, true);
```

### 自定义性能指标
```javascript
class PerformanceMonitor {
    constructor() {
        this.metrics = new Map();
        this.observer = new PerformanceObserver(this.handleEntries.bind(this));
        this.observer.observe({ entryTypes: ['measure', 'navigation', 'resource'] });
    }
    
    // 标记开始时间
    mark(name) {
        performance.mark(`${name}-start`);
    }
    
    // 测量耗时
    measure(name) {
        performance.mark(`${name}-end`);
        performance.measure(name, `${name}-start`, `${name}-end`);
    }
    
    handleEntries(list) {
        for (const entry of list.getEntries()) {
            if (entry.entryType === 'measure') {
                this.reportMetric({
                    name: entry.name,
                    duration: entry.duration,
                    startTime: entry.startTime
                });
            }
        }
    }
    
    reportMetric(metric) {
        fetch('/metrics', {
            method: 'POST',
            body: JSON.stringify(metric)
        });
    }
}

// 使用示例
const monitor = new PerformanceMonitor();

monitor.mark('api-call');
fetch('/api/data').then(() => {
    monitor.measure('api-call');
});
```

## 用户行为追踪

### 页面访问统计
```javascript
class PageTracker {
    constructor() {
        this.startTime = Date.now();
        this.setupListeners();
    }
    
    setupListeners() {
        // 页面可见性变化
        document.addEventListener('visibilitychange', () => {
            if (document.hidden) {
                this.trackPageLeave();
            } else {
                this.trackPageEnter();
            }
        });
        
        // 页面卸载
        window.addEventListener('beforeunload', () => {
            this.trackPageLeave();
        });
    }
    
    trackPageEnter() {
        this.startTime = Date.now();
        this.sendEvent('page_enter', {
            url: location.href,
            referrer: document.referrer
        });
    }
    
    trackPageLeave() {
        const duration = Date.now() - this.startTime;
        this.sendEvent('page_leave', {
            url: location.href,
            duration
        });
    }
    
    sendEvent(eventName, data) {
        navigator.sendBeacon('/analytics', JSON.stringify({
            event: eventName,
            data,
            timestamp: Date.now(),
            userId: this.getUserId()
        }));
    }
}
```

### 用户交互监控
```javascript
class InteractionTracker {
    constructor() {
        this.setupClickTracking();
        this.setupScrollTracking();
    }
    
    setupClickTracking() {
        document.addEventListener('click', (event) => {
            const element = event.target;
            const trackingData = {
                tagName: element.tagName,
                className: element.className,
                id: element.id,
                text: element.textContent?.slice(0, 100),
                xpath: this.getXPath(element)
            };
            
            this.track('click', trackingData);
        });
    }
    
    setupScrollTracking() {
        let maxScroll = 0;
        
        window.addEventListener('scroll', throttle(() => {
            const scrollPercent = Math.round(
                (window.scrollY / (document.body.scrollHeight - window.innerHeight)) * 100
            );
            
            if (scrollPercent > maxScroll) {
                maxScroll = scrollPercent;
                this.track('scroll', { percent: scrollPercent });
            }
        }, 1000));
    }
    
    getXPath(element) {
        if (element.id) return `//*[@id="${element.id}"]`;
        
        const parts = [];
        while (element && element.nodeType === Node.ELEMENT_NODE) {
            let index = 0;
            let sibling = element.previousSibling;
            
            while (sibling) {
                if (sibling.nodeType === Node.ELEMENT_NODE && 
                    sibling.nodeName === element.nodeName) {
                    index++;
                }
                sibling = sibling.previousSibling;
            }
            
            const tagName = element.nodeName.toLowerCase();
            const pathIndex = index > 0 ? `[${index + 1}]` : '';
            parts.unshift(tagName + pathIndex);
            
            element = element.parentNode;
        }
        
        return parts.length ? '/' + parts.join('/') : '';
    }
}
```

## 错误上报优化

### 采样和去重
```javascript
class ErrorReporter {
    constructor(options = {}) {
        this.sampleRate = options.sampleRate || 0.1; // 10%采样率
        this.errorCache = new Set();
        this.maxCacheSize = 100;
    }
    
    report(error) {
        // 采样控制
        if (Math.random() > this.sampleRate) {
            return;
        }
        
        // 错误去重
        const errorKey = this.getErrorKey(error);
        if (this.errorCache.has(errorKey)) {
            return;
        }
        
        this.errorCache.add(errorKey);
        
        // 缓存大小控制
        if (this.errorCache.size > this.maxCacheSize) {
            const firstKey = this.errorCache.values().next().value;
            this.errorCache.delete(firstKey);
        }
        
        this.sendError(error);
    }
    
    getErrorKey(error) {
        return `${error.message}-${error.filename}-${error.lineno}`;
    }
    
    sendError(error) {
        // 批量上报
        this.errorQueue = this.errorQueue || [];
        this.errorQueue.push(error);
        
        if (this.errorQueue.length >= 10) {
            this.flush();
        }
        
        // 定时上报
        if (!this.flushTimer) {
            this.flushTimer = setTimeout(() => {
                this.flush();
            }, 5000);
        }
    }
    
    flush() {
        if (this.errorQueue.length === 0) return;
        
        fetch('/errors', {
            method: 'POST',
            body: JSON.stringify(this.errorQueue)
        });
        
        this.errorQueue = [];
        clearTimeout(this.flushTimer);
        this.flushTimer = null;
    }
}
```

## 监控数据分析

### 错误聚合
```javascript
class ErrorAnalyzer {
    analyzeErrors(errors) {
        const analysis = {
            byType: this.groupByType(errors),
            byPage: this.groupByPage(errors),
            byBrowser: this.groupByBrowser(errors),
            timeline: this.generateTimeline(errors)
        };
        
        return analysis;
    }
    
    groupByType(errors) {
        return errors.reduce((acc, error) => {
            const type = error.type || 'javascript';
            acc[type] = (acc[type] || 0) + 1;
            return acc;
        }, {});
    }
    
    groupByPage(errors) {
        return errors.reduce((acc, error) => {
            const page = new URL(error.url).pathname;
            acc[page] = (acc[page] || 0) + 1;
            return acc;
        }, {});
    }
    
    generateTimeline(errors) {
        const timeline = {};
        
        errors.forEach(error => {
            const hour = new Date(error.timestamp).getHours();
            timeline[hour] = (timeline[hour] || 0) + 1;
        });
        
        return timeline;
    }
}
```
