🎯 一、JavaScript/TypeScript 基础

1. 闭包是什么？项目中如何应用？

参考答案： 闭包是函数能够访问并记住其词法作用域，即使函数在作用域外执行。 应用场景：

  • 模块化开发（实现私有变量）
  • 防抖节流函数
  • React Hooks 的实现原理
// 防抖函数就是闭包的典型应用
function debounce(fn, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

2. TypeScript 中 interface 和 type 的区别？

参考答案：

  • interface：主要用于定义对象形状，支持声明合并，更适合面向对象编程
  • type：可以定义任何类型（联合类型、元组等），不支持声明合并
// interface - 支持扩展和合并
interface User { name: string; }
interface User { age: number; } // 合并

// type - 更灵活的类型定义
type Status = 'pending' | 'success' | 'error';

2. 如何优化超大型列表的渲染性能？

参考答案：

// 虚拟滚动实现
class VirtualScroll {
  constructor(container, itemHeight, totalItems, renderItem) {
    this.container = container;
    this.itemHeight = itemHeight;
    this.totalItems = totalItems;
    this.renderItem = renderItem;
    
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight);
    this.startIndex = 0;
    
    this.setupVirtualScroll();
  }
  
  setupVirtualScroll() {
    // 监听滚动事件
    this.container.addEventListener('scroll', () => {
      this.handleScroll();
    });
    
    // 设置容器高度
    this.container.style.height = `${this.totalItems * this.itemHeight}px`;
    
    this.renderVisibleItems();
  }
  
  handleScroll() {
    const scrollTop = this.container.scrollTop;
    const newStartIndex = Math.floor(scrollTop / this.itemHeight);
    
    if (newStartIndex !== this.startIndex) {
      this.startIndex = newStartIndex;
      this.renderVisibleItems();
    }
  }
  
  renderVisibleItems() {
    const endIndex = Math.min(
      this.startIndex + this.visibleCount + 5, // 缓冲项
      this.totalItems
    );
    
    // 只渲染可见区域
    for (let i = this.startIndex; i < endIndex; i++) {
      this.renderItem(i, i - this.startIndex);
    }
  }
}

2. 如何实现前端代码的灰度发布？

// 灰度发布控制器
class GrayReleaseManager {
  constructor() {
    this.userGroups = new Map(); // 用户分组
    this.features = new Map();   // 功能开关
  }
  
  // 基于用户ID分桶
  isInGrayRelease(userId, featureName, percentage = 10) {
    const hash = this.hashCode(userId);
    const bucket = hash % 100;
    return bucket < percentage;
  }
  
  // 基于设备特征
  isInGrayReleaseByDevice(deviceInfo, rules) {
    return rules.some(rule => this.matchRule(deviceInfo, rule));
  }
  
  // 动态加载不同版本
  async loadFeature(featureName, userId) {
    const shouldGray = this.isInGrayRelease(userId, featureName);
    const version = shouldGray ? 'new' : 'stable';
    
    try {
      const module = await import(`./features/${featureName}/${version}.js`);
      return module.default;
    } catch (error) {
      // 降级到稳定版本
      const fallback = await import(`./features/${featureName}/stable.js`);
      return fallback.default;
    }
  }
}

2. 小程序如何实现预加载和缓存策略？

参考答案：

class PreloadManager {
  constructor() {
    this.cache = new Map();
    this.preloadQueue = [];
  }
  
  // 预加载关键数据
  preloadCriticalData() {
    const preloadList = [
      { url: '/api/userInfo', key: 'user' },
      { url: '/api/config', key: 'config' },
      { url: '/api/homeData', key: 'home' }
    ];
    
    preloadList.forEach(item => {
      this.preloadQueue.push(this.preloadItem(item));
    });
    
    return Promise.all(this.preloadQueue);
  }
  
  async preloadItem({ url, key }) {
    try {
      const data = await this.requestWithCache(url);
      this.cache.set(key, data);
      return data;
    } catch (error) {
      console.warn(`Preload failed: ${url}`, error);
    }
  }
  
  // 带缓存的请求
  async requestWithCache(url, options = {}) {
    const cacheKey = this.generateCacheKey(url);
    const cache = wx.getStorageSync(cacheKey);
    
    if (cache && !this.isCacheExpired(cache)) {
      return cache.data;
    }
    
    const freshData = await this.request(url, options);
    
    // 更新缓存
    wx.setStorageSync(cacheKey, {
      data: freshData,
      timestamp: Date.now(),
      ttl: options.ttl || 5 * 60 * 1000 // 5分钟
    });
    
    return freshData;
  }
}

1. 如何设计一个前端错误监控和自动修复系统？

参考答案：

class ErrorAutoRecovery {
  constructor() {
    this.errorPatterns = new Map();
    this.recoveryStrategies = new Map();
    this.setupErrorMonitoring();
  }
  
  setupErrorMonitoring() {
    // 监听错误
    window.addEventListener('error', (e) => {
      this.analyzeError(e.error);
    });
    
    // 监听 Promise 拒绝
    window.addEventListener('unhandledrejection', (e) => {
      this.analyzeError(e.reason);
    });
  }
  
  analyzeError(error) {
    const pattern = this.extractErrorPattern(error);
    const count = this.errorPatterns.get(pattern) || 0;
    
    if (count > 3) { // 同一错误出现3次
      this.tryAutoRecovery(pattern, error);
    }
    
    this.errorPatterns.set(pattern, count + 1);
  }
  
  tryAutoRecovery(pattern, error) {
    const strategy = this.recoveryStrategies.get(pattern);
    
    if (strategy) {
      // 执行恢复策略
      strategy.execute(error);
    } else {
      // 降级方案
      this.fallbackRecovery(error);
    }
  }
  
  // 组件级恢复 - 重新挂载组件
  async remountComponent(componentName) {
    const container = document.getElementById(componentName);
    if (container) {
      // 卸载组件
      ReactDOM.unmountComponentAtNode(container);
      
      // 重新挂载
      const newComponent = await this.loadComponent(componentName);
      ReactDOM.render(newComponent, container);
    }
  }
}

2. 如何设计一个支持 A/B 测试的前端数据驱动系统？

参考答案：

class ABTestSystem {
  constructor() {
    this.experiments = new Map();
    this.metrics = new Map();
    this.userContext = this.getUserContext();
  }
  
  // 定义实验
  defineExperiment(name, variants, metrics) {
    this.experiments.set(name, {
      variants,
      metrics,
      allocation: this.generateAllocation(variants.length)
    });
  }
  
  // 获取用户分组
  getVariant(experimentName) {
    const experiment = this.experiments.get(experimentName);
    if (!experiment) return null;
    
    const userBucket = this.getUserBucket(this.userContext.userId);
    const variantIndex = experiment.allocation[userBucket];
    
    return experiment.variants[variantIndex];
  }
  
  // 追踪指标
  track(experimentName, metricName, value) {
    const key = `${experimentName}:${metricName}`;
    if (!this.metrics.has(key)) {
      this.metrics.set(key, []);
    }
    this.metrics.get(key).push({
      value,
      timestamp: Date.now(),
      variant: this.getVariant(experimentName),
      userContext: this.userContext
    });
  }
  
  // 统计分析
  analyzeResults(experimentName) {
    const experiment = this.experiments.get(experimentName);
    const results = {};
    
    experiment.variants.forEach((variant, index) => {
      const variantMetrics = this.getVariantMetrics(experimentName, variant);
      results[variant] = {
        conversionRate: this.calculateConversionRate(variantMetrics),
        confidence: this.calculateConfidence(variantMetrics)
      };
    });
    
    return results;
  }
}
